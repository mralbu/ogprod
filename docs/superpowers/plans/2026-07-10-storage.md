# Storage (production store + entity geometry) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the approved spec `docs/superpowers/specs/2026-07-10-storage-design.md`: an entities schema (well points, field ring-fence polygons) as first-class connector output, a Parquet/GeoParquet store with upsert semantics and DuckDB-backed loaders, and an `ingest` CLI command.

**Architecture:** Parquet/GeoParquet files are the storage format (`data/store/production/source=<s>/year=<y>/data.parquet` + `data/store/entities/<s>.parquet`); DuckDB is only the query engine. Connectors gain `parse_entities`/`fetch_entities` mirroring the existing `parse`/`fetch` split (parse pure + offline-testable, fetch optional + network).

**Tech Stack:** pandas, geopandas (GeoParquet, shapely), duckdb, pyarrow, httpx, pytest. Tooling: `uv`, `just`, ruff, ty.

## Global Constraints

- Python `>=3.11` (pyproject `requires-python`).
- Use `uv` for all env/deps work (`uv add`, `uv run`); never pip.
- ruff line length 100; lint selectors E,W,F,I,UP,B,SIM. Run `just check` (lint + typecheck + tests) before each commit; `just fmt` to fix formatting.
- All volumes in m³ (`schema.CANONICAL_UNIT`); all geometry in EPSG:4326.
- Canonical production columns and upsert key are fixed by the spec: upsert on `(source, entity_id, period, product)`.
- Parsing must work offline (fixtures in `tests/data/` mirror real publisher formats); network code is injectable via `httpx.Client` and tested with `httpx.MockTransport`.
- Every parser validates its own output (`schema.validate` / `schema.validate_entities`).
- Commit after each task with a conventional-commit message.

```python
ENTITY_COLUMNS: tuple[str, ...] = (
    "source",  # connector name, e.g. "anp"
    "entity_type",  # "well" | "field"
    "entity_id",  # source-native identifier — joins to production
    "entity_name",  # human-readable name
    "field",  # field the entity belongs to
    "basin",
    "region",  # state / province / offshore area
    "operator",
    "status",  # lifecycle when reported (producing / shut-in / ...), source-native text
    "geometry",  # Point for wells, (Multi)Polygon ring fences for fields
)

_ENTITY_STRING_COLUMNS = tuple(c for c in ENTITY_COLUMNS if c != "geometry")

#: Geometry types each entity_type may carry (missing geometry is always allowed).
_GEOMETRY_TYPES: dict[str, set[str]] = {
    EntityType.WELL.value: {"Point"},
    EntityType.FIELD.value: {"Polygon", "MultiPolygon"},
}

#: CRS every entities frame must be in.
ENTITY_CRS = "EPSG:4326"


def empty_entities() -> gpd.GeoDataFrame:
    """An empty GeoDataFrame with the canonical entity columns."""
    frame = pd.DataFrame({column: pd.Series(dtype="string") for column in _ENTITY_STRING_COLUMNS})
    geometry = gpd.GeoSeries([], name="geometry", crs=ENTITY_CRS)
    return gpd.GeoDataFrame(frame, geometry=geometry)[list(ENTITY_COLUMNS)]


def validate_entities(entities: gpd.GeoDataFrame) -> gpd.GeoDataFrame:
    """Check that ``entities`` conforms to the canonical entity schema.

    Returns the frame unchanged so calls can be inlined; raises
    :class:`SchemaError` describing the first problem found.
    """
    if not isinstance(entities, gpd.GeoDataFrame):
        raise SchemaError("entities must be a geopandas.GeoDataFrame")
    missing = [column for column in ENTITY_COLUMNS if column not in entities.columns]
    if missing:
        raise SchemaError(f"missing entity columns: {missing}")
    if len(entities) == 0:
        return entities

    bad_types = set(entities["entity_type"].dropna().unique()) - {e.value for e in EntityType}
    if bad_types:
        raise SchemaError(f"invalid entity_type values: {sorted(bad_types)}")

    if entities.crs is None or entities.crs.to_epsg() != 4326:
        raise SchemaError(f"entities must be in {ENTITY_CRS}, got {entities.crs}")

    for entity_type, allowed in _GEOMETRY_TYPES.items():
        geoms = entities.loc[entities["entity_type"] == entity_type, "geometry"].dropna()
        bad_geoms = set(geoms.geom_type.unique()) - allowed
        if bad_geoms:
            raise SchemaError(
                f"{entity_type} geometry must be {sorted(allowed)}, got {sorted(bad_geoms)}"
            )
    return entities
```

```python
def build_entities(
    *,
    source: str,
    entity_type: schema.EntityType,
    entity_id: pd.Series,
    entity_name: pd.Series,
    geometry: gpd.GeoSeries | list,
    field: pd.Series | None = None,
    basin: pd.Series | None = None,
    region: pd.Series | None = None,
    operator: pd.Series | None = None,
    status: pd.Series | None = None,
    crs: Any = schema.ENTITY_CRS,
) -> gpd.GeoDataFrame:
    """Assemble a canonical entities frame; reprojects to EPSG:4326 and validates.

    All series must share an index. ``crs`` describes the *incoming* geometry;
    pass the publisher's CRS and the frame is reprojected.
    """
    na = pd.Series(pd.NA, index=entity_id.index, dtype="string")

    def text(series: pd.Series | None) -> pd.Series:
        return na if series is None else series.astype("string")

    entities = gpd.GeoDataFrame(
        {
            "source": source,
            "entity_type": entity_type.value,
            "entity_id": entity_id.astype("string"),
            "entity_name": entity_name.astype("string"),
            "field": text(field),
            "basin": text(basin),
            "region": text(region),
            "operator": text(operator),
            "status": text(status),
            "geometry": geometry,
        },
        geometry="geometry",
        crs=crs,
    )
    if entities.crs is None or entities.crs.to_epsg() != 4326:
        entities = entities.to_crs(schema.ENTITY_CRS)
    entities = entities[list(schema.ENTITY_COLUMNS)].reset_index(drop=True)
    return schema.validate_entities(entities)
```

```python
"""Persistent local store: Parquet/GeoParquet files queried through DuckDB.

The store is a directory of plain files — Parquet for production
(hive-partitioned by source and year), GeoParquet for entities (one file per
source). DuckDB is only the query engine (see ``load_production`` /
``sql``), never the storage format, so everything here is readable directly
with pandas/geopandas and servable from static hosting.

Publishers restate history, so production writes upsert on
``(source, entity_id, period, product)`` by rewriting the affected
``(source, year)`` partitions — never by appending.
"""

from __future__ import annotations

import os
from pathlib import Path

import pandas as pd

from ogprod import schema

#: Default store location, overridable via ``OGPROD_STORE`` or per call.
DEFAULT_ROOT = Path("data/store")
_ENV_VAR = "OGPROD_STORE"

_UPSERT_KEY = ["entity_id", "period", "product"]
_SORT_KEY = ["entity_id", "period", "product"]


def resolve_root(root: str | Path | None = None) -> Path:
    """The store root: explicit argument, else $OGPROD_STORE, else data/store."""
    if root is not None:
        return Path(root)
    return Path(os.environ.get(_ENV_VAR, DEFAULT_ROOT))


def _partition_path(root: Path, source: str, year: int) -> Path:
    return root / "production" / f"source={source}" / f"year={year}" / "data.parquet"


def _atomic_write_parquet(frame, path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(path.name + ".tmp")
    frame.to_parquet(tmp, index=False)
    os.replace(tmp, path)


def write_production(frame: pd.DataFrame, root: str | Path | None = None) -> list[Path]:
    """Upsert a canonical frame into the store; returns rewritten partition paths.

    Incoming rows replace stored rows with the same
    ``(source, entity_id, period, product)``; duplicate keys within ``frame``
    itself keep the last occurrence (matters for ANP, whose pré-sal file
    restates a subset of the mar file).
    """
    schema.validate(frame)
    frame = frame.drop_duplicates(subset=["source", *_UPSERT_KEY], keep="last")
    written = []
    base = resolve_root(root)
    for (source, year), group in frame.groupby(["source", frame["period"].dt.year]):
        path = _partition_path(base, str(source), int(year))
        merged = group
        if path.exists():
            existing = pd.read_parquet(path)
            incoming_keys = pd.MultiIndex.from_frame(group[_UPSERT_KEY])
            kept = existing[~pd.MultiIndex.from_frame(existing[_UPSERT_KEY]).isin(incoming_keys)]
            merged = pd.concat([kept, group], ignore_index=True)
        merged = merged.sort_values(_SORT_KEY, ignore_index=True)[list(schema.CANONICAL_COLUMNS)]
        _atomic_write_parquet(merged, path)
        written.append(path)
    return written
```

```python
    return root / "entities" / f"{source}.parquet"


def write_entities(entities: gpd.GeoDataFrame, root: str | Path | None = None) -> list[Path]:
    """Replace each source's entities file (GeoParquet); returns written paths.

    Entities are small (10³–10⁴ rows per source), so whole-file replacement
    per source beats merge logic.
    """
    schema.validate_entities(entities)
    base = resolve_root(root)
    written = []
    for source, group in entities.groupby("source"):
        path = _entities_file(base, str(source))
        group = group.reset_index(drop=True)[list(schema.ENTITY_COLUMNS)]
        _atomic_write_parquet(group, path)
        written.append(path)
    return written


def load_entities(
    *,
    source: str | None = None,
    entity_type: str | None = None,
    root: str | Path | None = None,
) -> gpd.GeoDataFrame:
    """Load entities from the store, optionally filtered by source/entity_type."""
    base = resolve_root(root) / "entities"
    if source is not None:
        files = [base / f"{source}.parquet"]
    elif base.exists():
        files = sorted(base.glob("*.parquet"))
    else:
        files = []
    files = [f for f in files if f.exists()]
    if not files:
        return schema.empty_entities()
    parts = [gpd.read_parquet(f) for f in files]
    entities = parts[0] if len(parts) == 1 else pd.concat(parts, ignore_index=True)
    if entity_type is not None:
        entities = entities[entities["entity_type"] == entity_type].reset_index(drop=True)
    return entities
```

```python
def _production_files(
    root: Path, source: str | None, start: pd.Timestamp | None, end: pd.Timestamp | None
) -> list[Path]:
    """Partition files matching the source/year pruning implied by the filters."""
    base = root / "production"
    if not base.exists():
        return []
    source_dirs = [base / f"source={source}"] if source else sorted(base.glob("source=*"))
    files = []
    for source_dir in source_dirs:
        for year_dir in sorted(source_dir.glob("year=*")):
            year = int(year_dir.name.split("=", 1)[1])
            if start is not None and year < start.year:
                continue
            if end is not None and year > end.year:
                continue
            path = year_dir / "data.parquet"
            if path.exists():
                files.append(path)
    return files


def _files_literal(files: list[Path]) -> str:
    quoted = ", ".join("'" + str(f).replace("'", "''") + "'" for f in files)
    return f"[{quoted}]"


def load_production(
    *,
    source: str | Iterable[str] | None = None,
    entity_type: str | Iterable[str] | None = None,
    entity_id: str | Iterable[str] | None = None,
    field: str | Iterable[str] | None = None,
    basin: str | Iterable[str] | None = None,
    region: str | Iterable[str] | None = None,
    operator: str | Iterable[str] | None = None,
    product: str | Iterable[str] | None = None,
    start: str | pd.Timestamp | None = None,
    end: str | pd.Timestamp | None = None,
    root: str | Path | None = None,
) -> pd.DataFrame:
    """Load canonical production rows, filtered with DuckDB pushdown.

    String filters accept one value or an iterable; ``start``/``end`` bound
    ``period`` inclusively. Filters combine with AND. Source and year filters
    additionally prune which partition files are read at all.
    """
    start = None if start is None else pd.Timestamp(start)
    end = None if end is None else pd.Timestamp(end)
    single_source = source if isinstance(source, str) else None
    files = _production_files(resolve_root(root), single_source, start, end)
    if not files:
        return schema.empty_frame()

    conditions, params = [], []
    string_filters = {
        "source": source,
        "entity_type": entity_type,
        "entity_id": entity_id,
        "field": field,
        "basin": basin,
        "region": region,
        "operator": operator,
        "product": product,
    }
    for column, value in string_filters.items():
        if value is None:
            continue
        values = [value] if isinstance(value, str) else [str(v) for v in value]
        conditions.append(f"{column} IN ({', '.join('?' for _ in values)})")
        params.extend(values)
    if start is not None:
        conditions.append("period >= ?")
        params.append(start.to_pydatetime())
    if end is not None:
        conditions.append("period <= ?")
        params.append(end.to_pydatetime())

    where = f" WHERE {' AND '.join(conditions)}" if conditions else ""
    query = (
        f"SELECT * FROM read_parquet({_files_literal(files)}){where} "
        "ORDER BY source, entity_id, period, product"
    )
    frame = duckdb.execute(query, params).df()
    frame["period"] = frame["period"].astype("datetime64[ns]")
    frame = frame.astype({column: "string" for column in schema._STRING_COLUMNS})
    return frame[list(schema.CANONICAL_COLUMNS)]


def sql(query: str, root: str | Path | None = None) -> pd.DataFrame:
    """Run a DuckDB query over the store.

    Two views are available: ``production`` (canonical columns) and
    ``entities`` (entity columns; ``geometry`` arrives as WKB bytes — use
    :func:`load_entities` when you want real geometries).
    """
    base = resolve_root(root)
    con = duckdb.connect()
    production_files = _production_files(base, None, None, None)
    if production_files:
        con.execute(
            f"CREATE VIEW production AS SELECT * FROM read_parquet({_files_literal(production_files)})"
        )
    else:
        con.register("production", schema.empty_frame())
    entities_dir = base / "entities"
    entity_files = sorted(entities_dir.glob("*.parquet")) if entities_dir.exists() else []
    if entity_files:
        con.execute(
            "CREATE VIEW entities AS SELECT * FROM "
            f"read_parquet({_files_literal(entity_files)}, union_by_name=true)"
        )
    else:
        con.register("entities", pd.DataFrame(schema.empty_entities().drop(columns="geometry")))
    return con.execute(query).df()
```

Note: `schema._STRING_COLUMNS` is private; add a module comment where used (`# canonical string columns; shared with schema by design`) or promote it — simplest correct move is to reference it as-is (same package). If ruff/ty complain, add `STRING_COLUMNS = _STRING_COLUMNS` alias in `schema.py` and use that.

```python
"""Command-line entry point.

``python -m ogprod fetch <source>``   — download raw files, print their paths.
``python -m ogprod ingest <source>``  — fetch + parse + upsert into the store.
"""

from __future__ import annotations

import argparse
import sys
from pathlib import Path

import ogprod
from ogprod import store


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="ogprod", description="Organize public oil & gas production data."
    )
    commands = parser.add_subparsers(dest="command", required=True)

    fetch = commands.add_parser(
        "fetch",
        help="download a source's raw files",
        description="Download a source's raw production files and print their paths.",
        epilog=(
            "Per-source options: sodir takes none (whole history); "
            "argentina takes --year; anp takes --year (plus --month for 2021-2023) "
            "or no flags to download the whole 2005-2023 archive."
        ),
    )
    ingest = commands.add_parser(
        "ingest",
        help="fetch, parse and upsert a source into the local store",
        description=(
            "Download a source's raw files, parse them, and upsert the result "
            "into the Parquet store. Prints the store files written."
        ),
    )
    for sub in (fetch, ingest):
        sub.add_argument("source", choices=ogprod.available())
        sub.add_argument("--year", type=int, help="production year (anp, argentina)")
        sub.add_argument("--month", type=int, help="production month (anp monthly files)")
        sub.add_argument(
            "--entities",
            action="store_true",
            help="operate on entity geometry (well coordinates / field ring fences)",
        )
    fetch.add_argument(
        "--dest", type=Path, default=Path("data"), help="download directory (default: data/)"
    )
    ingest.add_argument(
        "--dest",
        type=Path,
        default=Path("data/raw"),
        help="download directory (default: data/raw/)",
    )
    ingest.add_argument(
        "--store",
        type=Path,
        default=None,
        dest="store_root",
        help="store root (default: data/store/ or $OGPROD_STORE)",
    )
    return parser


def _fetch(args: argparse.Namespace, options: dict) -> list[Path]:
    if args.entities:
        return ogprod.fetch_entities(args.source, args.dest, **options)
    return ogprod.fetch(args.source, args.dest, **options)


def _ingest(args: argparse.Namespace, options: dict) -> list[Path]:
    paths = _fetch(args, options)
    written: list[Path] = []
    if args.entities:
        for path in paths:
            written += store.write_entities(
                ogprod.load_entities(args.source, path), root=args.store_root
            )
    else:
        # One write per raw file, in fetch order: later files upsert over
        # earlier ones (ANP's pré-sal file restates a subset of the mar file).
        for path in paths:
            written += store.write_production(
                ogprod.load(args.source, path), root=args.store_root
            )
    return written


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    options = {key: value for key in ("year", "month") if (value := getattr(args, key)) is not None}
    try:
        if args.command == "fetch":
            paths = _fetch(args, options)
        else:
            paths = _ingest(args, options)
    except (ValueError, TypeError, NotImplementedError) as error:
        print(f"error: {error}", file=sys.stderr)
        return 1
    for path in dict.fromkeys(paths):
        print(path)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
    def fetch_entities(
        self, dest_dir: str | Path, *, client: httpx.Client | None = None, **kwargs: Any
    ) -> list[Path]:
        """Download the FactMaps field-area shapefile zip (field ring fences)."""
        dest_dir = Path(dest_dir)
        dest_dir.mkdir(parents=True, exist_ok=True)
        with base.open_client(client) as http:
            return [base.download(http, _AREA_URL, dest_dir / "fldArea.zip")]

    def parse_entities(self, raw: base.RawFile) -> gpd.GeoDataFrame:
        frame = gpd.read_file(raw)
        columns = {column.lower(): column for column in frame.columns}
        roles: dict[str, str] = {}
        for alias, role in _ENTITY_ALIASES.items():
            if alias in columns and role not in roles:
                roles[role] = columns[alias]
        if "entity_name" not in roles or "entity_id" not in roles:
            raise ValueError(
                f"not a Sodir field-area file; columns found: {sorted(frame.columns)}"
            )

        # One geometry per field: FactMaps splits fields into several polygons.
        dissolved = frame.dissolve(by=roles["entity_id"], aggfunc="first").reset_index()
        name = dissolved[roles["entity_name"]].astype("string")

        def optional(role: str) -> pd.Series | None:
            return dissolved[roles[role]] if role in roles else None

        return base.build_entities(
            source=self.name,
            entity_type=self.entity_type,
            entity_id=dissolved[roles["entity_id"]].astype("string"),
            entity_name=name,
            geometry=dissolved.geometry,
            field=name,
            region=optional("region"),
            operator=optional("operator"),
            status=optional("status"),
            crs=dissolved.crs,
        )
```

```python
    def fetch_entities(
        self, dest_dir: str | Path, *, client: httpx.Client | None = None, **kwargs: Any
    ) -> list[Path]:
        """Download the well cadastre CSV (one row per drilled well, with coordinates)."""
        dest_dir = Path(dest_dir)
        dest_dir.mkdir(parents=True, exist_ok=True)
        with base.open_client(client) as http:
            response = http.get(_CADASTRE_PAGE_URL)
            response.raise_for_status()
            match = _CADASTRE_LINK.search(response.text)
            if match is None:
                raise ValueError(
                    f"no cadastre CSV link found on {_CADASTRE_PAGE_URL}; "
                    "download it manually and pass the path to parse_entities"
                )
            url = match.group("url")
            return [base.download(http, url, dest_dir / url.rsplit("/", 1)[-1])]

    def parse_entities(self, raw: base.RawFile) -> gpd.GeoDataFrame:
        frame = pd.read_csv(raw, sep=";", decimal=",", encoding="utf-8-sig", low_memory=False)
        normalized = {_normalize(column): column for column in frame.columns}

        meta: dict[str, pd.Series] = {}
        for alias, role in _ENTITY_META_ALIASES.items():
            if alias in normalized and role not in meta:
                meta[role] = frame[normalized[alias]]
        latitude = next((normalized[a] for a in _LATITUDE_ALIASES if a in normalized), None)
        longitude = next((normalized[a] for a in _LONGITUDE_ALIASES if a in normalized), None)
        if "entity_name" not in meta or latitude is None or longitude is None:
            raise ValueError(
                "not an ANP well cadastre file; needs a 'Poço' column and "
                f"decimal-degree coordinates, got: {sorted(frame.columns)}"
            )

        geometry = gpd.points_from_xy(
            pd.to_numeric(frame[longitude], errors="coerce"),
            pd.to_numeric(frame[latitude], errors="coerce"),
        )
        name = meta["entity_name"]
        return base.build_entities(
            source=self.name,
            entity_type=self.entity_type,
            entity_id=name,
            entity_name=name,
            geometry=geometry,
            field=meta.get("field"),
            basin=meta.get("basin"),
            region=meta.get("region"),
            operator=meta.get("operator"),
            status=meta.get("status"),
        )
```
